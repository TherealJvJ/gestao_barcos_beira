```python
import requests
from decouple import config

def enviar_sms(telefone, mensagem):
    url = "https://api.mozesms.com/sms/send"
    headers = {
        "X-API-Key": config('MOZESMS_API_KEY'),
        "X-API-Secret": config('MOZESMS_API_SECRET')
    }
    data = {"to": telefone, "message": mensagem, "from": config('MOZESMS_SENDER_ID')}
    return requests.post(url, json=data, headers=headers)
```
